# Publica o MoneyControl no GitHub Pages sempre que você envia mudanças para a branch "main".
# Também roda os testes de cálculo antes: se algum falhar, nada é publicado.
name: Publicar o app

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  publicar:
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'

      - name: Testes de cálculo
        run: python -m unittest discover -s tests

      - uses: actions/setup-node@v4
        with:
          node-version: 20

      - name: Montar o app (baixa o Pyodide e prepara os arquivos)
        run: node scripts/montar-site.mjs

      - uses: actions/configure-pages@v5

      - uses: actions/upload-pages-artifact@v3
        with:
          path: dist

      - id: deployment
        uses: actions/deploy-pages@v4
